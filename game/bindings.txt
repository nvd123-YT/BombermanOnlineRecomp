# Bomberman Online — default input bindings (dreamcomp format).
#
# This file is the port's documented default. The launcher copies it to
# .dreamcomp/bindings.txt on first run and passes that copy with --bindings, so rebinding in the
# window (F1) changes the user's copy rather than this tracked file.
#
# Physical input names are SDL's own names, so a layout survives an SDL upgrade. A control may
# have a second slot (".2") and either slot drives it.
#
# Dreamcast mapping used by the game: d-pad moves, A confirms/places a bomb, B cancels, START
# pauses. X/Y are also mapped so the binding screen and any menu that uses them work.

version = 1
deadzone = 20
trigger_ramp = 1

[keyboard]
a = key:Z
b = key:X
x = key:A
y = key:S
start = key:Return
start.2 = key:Space
dpad_up = key:Up
dpad_down = key:Down
dpad_left = key:Left
dpad_right = key:Right
ltrigger = key:Q
rtrigger = key:W
# A keyboard has no analogue stick, so the d-pad drives the stick too. Without this a title that
# steers with the stick and ignores the d-pad is unplayable from a keyboard.
stick_up = key:Up
stick_down = key:Down
stick_left = key:Left
stick_right = key:Right

[gamepad]
a = button:a
b = button:b
x = button:x
y = button:y
start = button:start
dpad_up = button:dpup
dpad_down = button:dpdown
dpad_left = button:dpleft
dpad_right = button:dpright
ltrigger = axis:lefttrigger+
rtrigger = axis:righttrigger+
stick_up = axis:lefty-
stick_down = axis:lefty+
stick_left = axis:leftx-
stick_right = axis:leftx+
