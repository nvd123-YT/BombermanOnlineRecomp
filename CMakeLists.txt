cmake_minimum_required(VERSION 3.24)
project(bomberman_online_recomp VERSION 0.1.0 LANGUAGES C CXX)

# Bomberman Online (USA, MK-51065 V1.004) as a native PC port, built with dreamcomp.
# dreamcomp is expected next to this repository or at $DREAMCOMP_DIR / ~/Tools/dreamcomp.
set(DREAMCOMP_DIR "" CACHE PATH "dreamcomp checkout")
if(NOT DREAMCOMP_DIR)
  if(DEFINED ENV{DREAMCOMP_DIR} AND EXISTS "$ENV{DREAMCOMP_DIR}/cmake/DreamcompPort.cmake")
    set(DREAMCOMP_DIR "$ENV{DREAMCOMP_DIR}")
  elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dreamcomp/cmake/DreamcompPort.cmake")
    set(DREAMCOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/dreamcomp")
  elseif(EXISTS "$ENV{HOME}/Tools/dreamcomp/cmake/DreamcompPort.cmake")
    set(DREAMCOMP_DIR "$ENV{HOME}/Tools/dreamcomp")
  else()
    message(FATAL_ERROR "dreamcomp not found: pass -DDREAMCOMP_DIR=<checkout>")
  endif()
endif()
message(STATUS "dreamcomp: ${DREAMCOMP_DIR}")
add_subdirectory(${DREAMCOMP_DIR} dreamcomp)
add_subdirectory(game)
